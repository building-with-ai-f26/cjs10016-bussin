# Bussin iOS Phase 1 — Walking Skeleton Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** A real iOS app showing NYU's seven shuttle routes, their stops, and live buses moving on Apple Maps, polling the Passio feed every 4 seconds — plus a separate ridership-collector repo logging `paxLoad` around the clock.

**Architecture:** Pure SwiftUI + MapKit `Map`, iOS 18 minimum. One `@Observable` store (`ShuttleStore`) owns all data; a `PassioFetching` protocol separates networking from parsing so every parser is tested against recorded real-feed JSON. The Xcode project is generated from `ios/project.yml` by XcodeGen (dev tool only — not an app dependency).

**Tech Stack:** Swift (language mode 5), SwiftUI, MapKit, XCTest, XcodeGen, PostHog iOS SDK (optional at runtime), GitHub Actions (collector).

**Context for the executor:**
- Spec: `docs/superpowers/specs/2026-11-09-bussin-ios-app-design.md`. Read it first.
- The working tree already holds unrelated uncommitted changes (prototype bus view, PRD edits). **`git add` only the files each task names — never `git add -A`.**
- Feed facts verified live on Nov 9: `getStops` returns `stops` keyed `"ID<stopId>"` with numeric lat/lng; `routes` keyed by routeId with value `[name, color, [pos, stopId, 0], …]`; `routePoints` keyed by routeId with value = array of segments, each segment an array of `{lat, lng}` **strings**; `routeShortNames` maps routeId → letter (nullable). `getBuses` returns `buses` keyed by deviceId, each value an **array** holding one record; lat/lng and `calculatedCourse` are strings; `paxLoad`/`totalCap` are ints. NYU Langone commuter routes share `userId` "1007", so filter by lettered routes, not user id. When no buses run, PHP may emit `[]` instead of `{}` — every top-level decode needs a fallback.
- Simulator: "iPhone 17" exists on this machine. Xcode 27.2 at `/Applications/Xcode.app`.
- Model: this phase runs on Fable/Opus.

`ios/Bussin/Networking/PassioPayloads.swift`:

```swift
import Foundation
import CoreLocation

/// Raw shapes of passio3.com/www/mapGetData.php responses.
/// The feed is PHP: empty collections may arrive as `[]` instead of `{}`,
/// numbers may arrive as strings, and some values are null. Every decode
/// here tolerates that rather than throwing the whole payload away.

struct PassioStopsPayload: Decodable {
    var stops: [String: PassioStop] = [:]            // keyed "ID<stopId>"
    var routes: [String: PassioRouteEntry] = [:]     // keyed by routeId
    var routePoints: [String: [[PassioPoint]]] = [:] // routeId -> segments -> points
    var routeShortNames: [String: String] = [:]      // routeId -> letter ("A"); nulls dropped
    var excludedRoutesID: [Int] = []                 // routes Passio has suspended right now

    private enum CodingKeys: String, CodingKey {
        case stops, routes, routePoints, routeShortNames, excludedRoutesID
    }

    init(from decoder: Decoder) throws {
        let c = try decoder.container(keyedBy: CodingKeys.self)
        stops = (try? c.decode([String: PassioStop].self, forKey: .stops)) ?? [:]
        routes = (try? c.decode([String: PassioRouteEntry].self, forKey: .routes)) ?? [:]
        routePoints = (try? c.decode([String: [[PassioPoint]]].self, forKey: .routePoints)) ?? [:]
        let names = (try? c.decode([String: String?].self, forKey: .routeShortNames)) ?? [:]
        routeShortNames = names.compactMapValues { $0 }
        excludedRoutesID = (try? c.decode([Int].self, forKey: .excludedRoutesID)) ?? []
    }
}

struct PassioStop: Decodable {
    let stopId: String
    let name: String
    let latitude: Double
    let longitude: Double
}

struct PassioPoint: Decodable {
    let lat: String
    let lng: String
    var coordinate: CLLocationCoordinate2D {
        .init(latitude: Double(lat) ?? 0, longitude: Double(lng) ?? 0)
    }
}

/// Feed shape: ["Route A", "#f7b6d2", ["1", "6545", 0], ["2", "6574", 0], ...]
struct PassioRouteEntry: Decodable {
    let name: String
    let color: String
    let stopIds: [String]

    init(from decoder: Decoder) throws {
        var c = try decoder.unkeyedContainer()
        name = try c.decode(String.self)
        color = try c.decode(String.self)
        var ids: [String] = []
        while !c.isAtEnd {
            if let ref = try? c.decode(PassioStopRef.self) {
                ids.append(ref.stopId)
            } else if (try? c.decode(String.self)) != nil || (try? c.decode(Double.self)) != nil {
                continue  // consumed an unexpected scalar; move on
            } else {
                break     // unknown shape — stop rather than loop forever
            }
        }
        stopIds = ids
    }
}

struct PassioStopRef: Decodable {
    let stopId: String
    init(from decoder: Decoder) throws {
        var c = try decoder.unkeyedContainer()
        _ = try c.decode(String.self)        // position ("1", "2", ...)
        stopId = try c.decode(String.self)   // the stop id
    }
}
```

`ios/Bussin/Models/ShuttleModels.swift`:

```swift
import Foundation
import CoreLocation

struct ShuttleRoute: Identifiable {
    let id: String                              // Passio routeId, e.g. "74769"
    let name: String                            // "Route A"
    let letter: String                          // "A"
    let colorHex: String                        // "#f7b6d2"
    let coordinates: [CLLocationCoordinate2D]   // full polyline, segments flattened in order
    let stopIds: [String]                       // boarding order; loops repeat the first stop last
}

struct ShuttleStop: Identifiable {
    let id: String
    let name: String
    let coordinate: CLLocationCoordinate2D
    let routeIds: [String]
}

enum ShuttleLetters {
    static let allowed: Set<String> = ["A", "B", "C", "E", "F", "G", "W"]
}

struct ShuttleCatalog {
    let routes: [ShuttleRoute]
    let stops: [ShuttleStop]
    let excludedRouteIds: Set<String>

    static func from(payload: PassioStopsPayload) -> ShuttleCatalog {
        var routes: [ShuttleRoute] = []
        var stopRouteIds: [String: [String]] = [:]

        for (rid, entry) in payload.routes {
            guard let letter = payload.routeShortNames[rid],
                  ShuttleLetters.allowed.contains(letter) else { continue }
            let pts = (payload.routePoints[rid] ?? []).flatMap { $0 }.map(\.coordinate)
            guard !pts.isEmpty else { continue }
            routes.append(ShuttleRoute(id: rid, name: entry.name, letter: letter,
                                       colorHex: entry.color, coordinates: pts,
                                       stopIds: entry.stopIds))
            for sid in Set(entry.stopIds) {
                stopRouteIds[sid, default: []].append(rid)
            }
        }
        routes.sort { $0.letter < $1.letter }

        var stops: [ShuttleStop] = []
        for (sid, rids) in stopRouteIds {
            guard let raw = payload.stops["ID" + sid] else { continue }
            stops.append(ShuttleStop(id: sid, name: raw.name,
                                     coordinate: .init(latitude: raw.latitude, longitude: raw.longitude),
                                     routeIds: rids.sorted()))
        }
        stops.sort { $0.name < $1.name }

        return ShuttleCatalog(routes: routes, stops: stops,
                              excludedRouteIds: Set(payload.excludedRoutesID.map(String.init)))
    }
}
```

```swift
import Foundation
import CoreLocation

struct LiveBus: Identifiable {
    let id: Int                     // Passio deviceId — stable per physical bus
    let number: String              // painted fleet number, e.g. "2146"
    let routeId: String
    let coordinate: CLLocationCoordinate2D
    let headingDegrees: Double?     // 0 = north
    let ridersAboard: Int?          // paxLoad
    let capacity: Int?              // totalCap
    let isStale: Bool               // Passio's own outdated flag

    static func from(payload: PassioBusesPayload, allowedRouteIds: Set<String>) -> [LiveBus] {
        payload.buses.values.flatMap { $0 }.compactMap { raw in
            guard let rid = raw.routeId, allowedRouteIds.contains(rid),
                  raw.outOfService != 1,
                  let lat = Double(raw.latitude), let lng = Double(raw.longitude)
            else { return nil }
            return LiveBus(id: raw.deviceId,
                           number: raw.bus ?? String(raw.deviceId),
                           routeId: rid,
                           coordinate: .init(latitude: lat, longitude: lng),
                           headingDegrees: raw.calculatedCourse.flatMap(Double.init),
                           ridersAboard: raw.paxLoad,
                           capacity: raw.totalCap,
                           isStale: raw.outdated == 1)
        }
        .sorted { $0.id < $1.id }
    }
}
```

```swift
import Foundation

protocol PassioFetching: Sendable {
    func stopsData() async throws -> Data
    func busesData() async throws -> Data
}

/// Thin client for passio3.com. No auth; the payload rides in a form-encoded `json` field.
struct PassioAPI: PassioFetching {
    static func url(query: String) -> URL {
        URL(string: "https://passio3.com/www/mapGetData.php?" + query)!
    }

    static func formBody(json: String) -> Data {
        var allowed = CharacterSet.alphanumerics
        allowed.insert(charactersIn: "-._~")
        let encoded = json.addingPercentEncoding(withAllowedCharacters: allowed) ?? json
        return Data("json=\(encoded)".utf8)
    }

    private func post(query: String, json: String) async throws -> Data {
        var req = URLRequest(url: Self.url(query: query))
        req.httpMethod = "POST"
        req.setValue("application/x-www-form-urlencoded", forHTTPHeaderField: "Content-Type")
        req.httpBody = Self.formBody(json: json)
        req.timeoutInterval = 10
        let (data, resp) = try await URLSession.shared.data(for: req)
        guard let http = resp as? HTTPURLResponse, http.statusCode == 200 else {
            throw URLError(.badServerResponse)
        }
        return data
    }

    func stopsData() async throws -> Data {
        try await post(query: "getStops=2", json: #"{"s0":"1007","sA":1}"#)
    }

    func busesData() async throws -> Data {
        try await post(query: "getBuses=2", json: #"{"s0":"1007","sA":1}"#)
    }
}
```

```swift
import Foundation
import Observation

@Observable @MainActor
final class ShuttleStore {
    enum FeedState: Equatable {
        case loading
        case live
        case stale(String)
    }

    private(set) var routes: [ShuttleRoute] = []
    private(set) var stops: [ShuttleStop] = []
    private(set) var buses: [LiveBus] = []
    private(set) var excludedRouteIds: Set<String> = []
    private(set) var feedState: FeedState = .loading

    var staleAfter: TimeInterval = 30
    private var lastBusUpdate: Date?
    private let api: any PassioFetching
    private var routesById: [String: ShuttleRoute] = [:]

    init(api: any PassioFetching = PassioAPI()) {
        self.api = api
    }

    func route(for id: String) -> ShuttleRoute? { routesById[id] }

    // MARK: - Pure appliers (testable without networking)

    func apply(catalogData: Data) {
        guard let payload = try? JSONDecoder().decode(PassioStopsPayload.self, from: catalogData) else { return }
        let catalog = ShuttleCatalog.from(payload: payload)
        guard !catalog.routes.isEmpty else { return }   // never let garbage wipe a good catalog
        routes = catalog.routes
        stops = catalog.stops
        excludedRouteIds = catalog.excludedRouteIds
        routesById = Dictionary(uniqueKeysWithValues: catalog.routes.map { ($0.id, $0) })
    }

    func apply(busesData data: Data) {
        guard let payload = try? JSONDecoder().decode(PassioBusesPayload.self, from: data) else { return }
        buses = LiveBus.from(payload: payload, allowedRouteIds: Set(routesById.keys))
        lastBusUpdate = .now
        feedState = .live
    }

    // MARK: - Feed access

    func loadCatalogFromSnapshot() {
        guard routes.isEmpty,
              let url = Bundle.main.url(forResource: "passio-stops", withExtension: "json"),
              let data = try? Data(contentsOf: url) else { return }
        apply(catalogData: data)
    }

    func refreshCatalog() async {
        if let data = try? await api.stopsData() { apply(catalogData: data) }
    }

    func pollOnce() async {
        do {
            apply(busesData: try await api.busesData())
        } catch {
            let age = lastBusUpdate.map { Date.now.timeIntervalSince($0) } ?? .infinity
            if age >= staleAfter {
                feedState = .stale("Live feed unreachable — showing last known buses")
            }
        }
    }

    /// Call from `.task` on the root view; cancels with the view.
    func run() async {
        loadCatalogFromSnapshot()
        await refreshCatalog()
        while !Task.isCancelled {
            await pollOnce()
            try? await Task.sleep(for: .seconds(4))
        }
    }
}
```

```swift
import SwiftUI

extension Color {
    /// "#f7b6d2" or "f7b6d2" → Color. Falls back to gray on junk.
    init(hex: String) {
        var h = hex.trimmingCharacters(in: CharacterSet.alphanumerics.inverted)
        if h.count == 3 { h = h.map { "\($0)\($0)" }.joined() }
        let v = UInt64(h, radix: 16) ?? 0x8E8E93
        self.init(red: Double((v >> 16) & 0xFF) / 255,
                  green: Double((v >> 8) & 0xFF) / 255,
                  blue: Double(v & 0xFF) / 255)
    }

    /// Black or white text, whichever reads on the given background.
    static func textOn(hex: String) -> Color {
        var h = hex.trimmingCharacters(in: CharacterSet.alphanumerics.inverted)
        if h.count == 3 { h = h.map { "\($0)\($0)" }.joined() }
        let v = UInt64(h, radix: 16) ?? 0x8E8E93
        let r = Double((v >> 16) & 0xFF), g = Double((v >> 8) & 0xFF), b = Double(v & 0xFF)
        let luma = (0.299 * r + 0.587 * g + 0.114 * b) / 255
        return luma > 0.6 ? .black : .white
    }
}
```

```python
#!/usr/bin/env python3
"""Append one JSONL row per active NYU shuttle to data/YYYY-MM-DD.jsonl (UTC).

Run by GitHub Actions every ~15 minutes. Passio's getBuses feed includes
paxLoad (riders aboard, from the bus's own counter) and totalCap per bus.
"""
import datetime
import json
import pathlib
import urllib.parse
import urllib.request

URL = "https://passio3.com/www/mapGetData.php?getBuses=2"
LETTERED = {"74768", "74769", "74770", "74771", "74772", "74773", "74774"}

body = urllib.parse.urlencode({"json": json.dumps({"s0": "1007", "sA": 1})}).encode()
req = urllib.request.Request(URL, data=body,
                             headers={"Content-Type": "application/x-www-form-urlencoded"})
raw = json.load(urllib.request.urlopen(req, timeout=30))

buses = raw.get("buses") or {}
if not isinstance(buses, dict):   # PHP emits [] when empty
    buses = {}

now = datetime.datetime.now(datetime.timezone.utc)
rows = []
for group in buses.values():
    for b in group:
        if b.get("routeId") not in LETTERED:
            continue
        rows.append({
            "ts": now.isoformat(timespec="seconds"),
            "routeId": b.get("routeId"),
            "route": b.get("route"),
            "bus": b.get("bus"),
            "riders": b.get("paxLoad"),
            "capacity": b.get("totalCap"),
            "lat": b.get("latitude"),
            "lng": b.get("longitude"),
            "outOfService": b.get("outOfService"),
            "outdated": b.get("outdated"),
        })

out = pathlib.Path("data") / f"{now:%Y-%m-%d}.jsonl"
out.parent.mkdir(exist_ok=True)
with out.open("a") as f:
    for r in rows:
        f.write(json.dumps(r) + "\n")
print(f"appended {len(rows)} rows to {out}")
```
